@app.post("/download-comparison-json") 
async def download_comparison_json(comparison: ReportComparison):
    """Télécharge la comparaison de deux rapports en JSON"""
    conn = None
    
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        # Validation des IDs
        if not comparison.report1_id or not comparison.report2_id:
            raise HTTPException(status_code=400, detail="IDs de rapports manquants")
        
        if comparison.report1_id == comparison.report2_id:
            raise HTTPException(status_code=400, detail="Impossible de comparer un rapport avec lui-même")
        
        print(f"🔍 Comparaison demandée: {comparison.report1_id} vs {comparison.report2_id}")
        
        # Récupérer les deux rapports avec vérification d'existence
        cursor.execute("SELECT id, filename, upload_date FROM reports WHERE id = ?", (comparison.report1_id,))
        report1 = cursor.fetchone()
        
        cursor.execute("SELECT id, filename, upload_date FROM reports WHERE id = ?", (comparison.report2_id,))
        report2 = cursor.fetchone()
        
        if not report1:
            raise HTTPException(status_code=404, detail=f"Rapport {comparison.report1_id} introuvable")
        if not report2:
            raise HTTPException(status_code=404, detail=f"Rapport {comparison.report2_id} introuvable")
        
        print("✅ Rapports trouvés en base")
        
        # Récupérer les textes pour l'analyse
        cursor.execute("SELECT extracted_text FROM reports WHERE id = ?", (comparison.report1_id,))
        report1_data = cursor.fetchone()
        report1_text = report1_data["extracted_text"] if report1_data else ""
        
        cursor.execute("SELECT extracted_text FROM reports WHERE id = ?", (comparison.report2_id,))
        report2_data = cursor.fetchone()
        report2_text = report2_data["extracted_text"] if report2_data else ""
        
        # Validation des textes
        if not report1_text or not report2_text:
            raise HTTPException(status_code=400, detail="Texte extrait manquant dans un ou plusieurs rapports")
        
        # Optimiser la taille des textes (réduire pour économiser les tokens)

        print(f"📝 Textes préparés: R1={len(report1_preview)} chars, R2={len(report2_preview)} chars")
        
        # Générer la comparaison avec timeout et retry
        comparison_analysis = await generate_comparison_analysis(
            report1, report2, report1_preview, report2_preview
        )
        
        print("✅ Analyse générée par IA")
        
        # Créer l'objet JSON avec validation
        json_data = {
            "type": "comparaison",
            "rapport1": {
                "id": report1["id"],
                "filename": report1["filename"],
                "upload_date": report1["upload_date"],
                "text_length": len(report1_text)
            },
            "rapport2": {
                "id": report2["id"],
                "filename": report2["filename"],
                "upload_date": report2["upload_date"],
                "text_length": len(report2_text)
            },
            "analyse_comparative": comparison_analysis,
            "metadata": {
                "date_generation": datetime.now().isoformat(),
                "modele_ia": OPENROUTER_MODEL,
                "version": "1.1",
                "text_analyzed_length": {
                    "rapport1": len(report1_preview),
                    "rapport2": len(report2_preview)
                }
            }
        }
        
        # Valider que le JSON est sérialisable
        try:
            json_content = json.dumps(json_data, ensure_ascii=False, indent=2)
        except TypeError as e:
            print(f"❌ Erreur sérialisation JSON: {e}")
            raise HTTPException(status_code=500, detail="Erreur de sérialisation des données")
        
        filename = f"comparaison_esg_{comparison.report1_id}_vs_{comparison.report2_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"

async def generate_comparison_analysis(report1, report2, report1_text, report2_text):
    """Génère l'analyse comparative avec gestion d'erreur robuste"""
    
    prompt = f"""
    Compare ces 2 rapports ESG selon les critères environnementaux ESRS (E1 à E5) :

    **RAPPORT 1 - {report1["filename"]}:**
    {report1_text}

    **RAPPORT 2 - {report2["filename"]}:**
    {report2_text}

    Analyse structurée :
    1. **SYNTHÈSE ENVIRONNEMENTALE** (2 lignes)
    2. **POINTS COMMUNS** (E1 à E5)
    3. **DIFFÉRENCES CLÉS** en matière environnementale
    4. **FORCES/FAIBLESSES** de chaque rapport (pilier E)
    5. **RECOMMANDATIONS** pour s’aligner aux ESRS
    6. **SCORE ENVIRONNEMENTAL** sur 10

    Réponse concise, professionnelle et en français.
    """

    
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json"
    }
    
    payload = {
        "model": OPENROUTER_MODEL,
        "messages": [
            {"role": "system", "content": "Tu es un expert en reporting ESG selon les normes ESRS qui effectue des analyses comparatives détaillées."},
            {"role": "user", "content": prompt}
        ],
        "max_tokens": 2800,  # Réduit pour rester sous la limite de crédits
        "temperature": 0.3
    }
    
    # Retry avec backoff exponentiel
    max_retries = 3
    for attempt in range(max_retries):
        try:
            timeout = httpx.Timeout(60.0)  # 60 secondes timeout
            async with httpx.AsyncClient(timeout=timeout) as client:
                print(f"🤖 Appel IA (tentative {attempt + 1}/{max_retries})...")
                response = await client.post(
                    "https://openrouter.ai/api/v1/chat/completions", 
                    json=payload, 
                    headers=headers
                )
                
                if response.status_code == 200:
                    result = response.json()
                    return result["choices"][0]["message"]["content"]
                elif response.status_code == 429:  # Rate limit
                    if attempt < max_retries - 1:
                        wait_time = 2 ** attempt  # Backoff exponentiel
                        print(f"⏳ Rate limit atteint, attente {wait_time}s...")
                        await asyncio.sleep(wait_time)
                        continue
                    else:
                        raise HTTPException(status_code=429, detail="Limite de taux API atteinte")
                else:
                    print(f"❌ Erreur API: {response.status_code} - {response.text}")
                    raise HTTPException(status_code=500, detail=f"Erreur API OpenRouter: {response.status_code}")
                    
        except httpx.TimeoutException:
            if attempt < max_retries - 1:
                print(f"⏳ Timeout, retry dans {2 ** attempt}s...")
                await asyncio.sleep(2 ** attempt)
                continue
            else:
                raise HTTPException(status_code=504, detail="Timeout API - réessayez plus tard")
        except Exception as e:
            if attempt < max_retries - 1:
                print(f"❌ Erreur réseau (tentative {attempt + 1}): {e}")
                await asyncio.sleep(2 ** attempt)
                continue
            else:
                raise HTTPException(status_code=500, detail=f"Erreur réseau: {str(e)}")
    
    raise HTTPException(status_code=500, detail="Impossible de générer l'analyse après plusieurs tentatives")


def generate_comparison_pdf(report1, report2, analysis):
    """Génère un PDF de comparaison"""
    try:
        buffer = io.BytesIO()
        doc = SimpleDocTemplate(buffer, pagesize=A4)
        styles = getSampleStyleSheet()
        
        # Style personnalisé
        title_style = ParagraphStyle(
            'CustomTitle',
            parent=styles['Heading1'],
            fontSize=16,
            spaceAfter=30,
            textColor='darkblue'
        )
        
        content = []
        
        # Titre
        title = f"Comparaison ESG - {report1['filename']} vs {report2['filename']}"
        content.append(Paragraph(title, title_style))
        content.append(Spacer(1, 12))
        
        # Informations des rapports
        info_text = f"""
        <b>Rapport 1:</b> {report1['filename']} (ID: {report1['id']})<br/>
        <b>Date upload:</b> {report1['upload_date']}<br/><br/>
        
        <b>Rapport 2:</b> {report2['filename']} (ID: {report2['id']})<br/>
        <b>Date upload:</b> {report2['upload_date']}<br/><br/>
        
        <b>Date de comparaison:</b> {datetime.now().strftime('%d/%m/%Y %H:%M')}
        """
        content.append(Paragraph(info_text, styles['Normal']))
        content.append(Spacer(1, 20))
        
        # Analyse (nettoyer le texte pour PDF)
        clean_analysis = analysis.replace('**', '<b>').replace('**', '</b>')
        clean_analysis = clean_analysis.replace('➤', '•')
        
        # Diviser l'analyse en paragraphes
        paragraphs = clean_analysis.split('\n\n')
        for para in paragraphs:
            if para.strip():
                content.append(Paragraph(para.strip(), styles['Normal']))
                content.append(Spacer(1, 10))
        
        # Construire le PDF
        doc.build(content)
        pdf_bytes = buffer.getvalue()
        buffer.close()
        
        return pdf_bytes
        
    except Exception as e:
        print(f"❌ Erreur génération PDF: {e}")
        raise HTTPException(status_code=500, detail=f"Erreur génération PDF: {str(e)}")

@app.post("/download-comparison-json")
async def download_comparison_json(comparison: ReportComparison):
    """
    Télécharge une comparaison au format JSON
    """
    try:
        cached = get_existing_comparison(comparison.user_id, comparison.report1_id, comparison.report2_id)
        if not cached:
            raise HTTPException(status_code=404, detail="Aucune comparaison trouvée")
        
        return Response(
            content=json.dumps(cached, ensure_ascii=False, indent=2),
            media_type="application/json",
            headers={
                "Content-Disposition": f"attachment; filename=comparaison_{comparison.report1_id}_vs_{comparison.report2_id}.json"
            }
        )
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erreur téléchargement JSON: {str(e)}")
def debug_database_tables():

def get_analysis_history_fixed(user_id: int = None, report_id: int = None, limit: int = 100):
    """
    Version corrigée qui cherche dans toutes les tables possibles
    """
    conn = get_db_connection()
    cursor = conn.cursor()
    
    try:
        # Lister toutes les tables
        cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
        all_tables = [row[0] for row in cursor.fetchall()]
        
        # Chercher les tables d'analyse possibles (plus large)
        possible_tables = [
            'advanced_analyses',
            'advanced_analysis', 
            'analyses',
            'analysis',
            'ai_analyses',
            'esg_analyses'
        ]
        
        # Trouver quelle table existe réellement
        existing_table = None
        for table in possible_tables:
            if table in all_tables:
                existing_table = table
                break
        
        # Si aucune table standard trouvée, chercher toute table contenant 'analy'
        if not existing_table:
            for table in all_tables:
                if 'analy' in table.lower():
                    existing_table = table
                    break
        
        if not existing_table:
            print(f"❌ Aucune table d'analyse trouvée dans: {all_tables}")
            return []
        
        print(f"✅ Utilisation de la table: {existing_table}")
        
        # Vérifier le nombre d'enregistrements
        cursor.execute(f"SELECT COUNT(*) FROM {existing_table}")
        total_count = cursor.fetchone()[0]
        print(f"📊 Total d'enregistrements dans {existing_table}: {total_count}")
        
        # Construire la requête
        base_query = f"""
            SELECT a.*, r.filename
            FROM {existing_table} a
            LEFT JOIN reports r ON a.report_id = r.id
        """
        
        conditions = []
        params = []
        
        if user_id is not None:
            conditions.append("a.user_id = ?")
            params.append(user_id)
            print(f"🔍 Filtrage par user_id: {user_id}")
        
        if report_id is not None:
            conditions.append("a.report_id = ?")
            params.append(report_id)
            print(f"🔍 Filtrage par report_id: {report_id}")
        
        if conditions:
            base_query += " WHERE " + " AND ".join(conditions)
        
        base_query += f" ORDER BY a.id DESC LIMIT {limit}"
        
        print(f"🔍 Requête SQL: {base_query}")
        print(f"🔍 Paramètres: {params}")
        
        cursor.execute(base_query, params)
        rows = cursor.fetchall()
        
        print(f"📊 Lignes trouvées: {len(rows)}")
        
        if len(rows) == 0 and conditions:
            # Test sans filtres pour voir s'il y a des données
            cursor.execute(f"SELECT COUNT(*) FROM {existing_table}")
            total = cursor.fetchone()[0]
            print(f"⚠️ Aucun résultat avec filtres, mais {total} enregistrements au total")
            
            # Montrer quelques IDs d'utilisateurs existants
            cursor.execute(f"SELECT DISTINCT user_id FROM {existing_table} LIMIT 5")
            existing_users = [row[0] for row in cursor.fetchall()]
            print(f"👥 IDs utilisateurs existants: {existing_users}")
        
        results = []
        for row in rows:
            try:
                result = {
                    "id": row["id"] if "id" in row.keys() else None,
                    "report_id": row["report_id"] if "report_id" in row.keys() else None,
                    "user_id": row["user_id"] if "user_id" in row.keys() else None,
                    "filename": row["filename"] if "filename" in row.keys() else f"Rapport {row['report_id'] if 'report_id' in row.keys() else 'N/A'}",
                    "analysis_text": row["analysis_text"] if "analysis_text" in row.keys() else "",
                    "model_used": row["model_used"] if "model_used" in row.keys() else None,
                    "tokens_used": row["tokens_used"] if "tokens_used" in row.keys() else 0,
                }
                
                # Date
                if "analysis_date" in row.keys():
                    result["analysis_date"] = row["analysis_date"]
                elif "created_at" in row.keys():
                    result["analysis_date"] = row["created_at"]
                else:
                    result["analysis_date"] = None
                
                # Scores
                scores = {}
                if "scores_json" in row.keys() and row["scores_json"]:
                    try:
                        scores = json.loads(row["scores_json"])
                    except:
                        pass
                elif "scores" in row.keys() and row["scores"]:
                    try:
                        scores = json.loads(row["scores"]) if isinstance(row["scores"], str) else row["scores"]
                    except:
                        pass
                
                result["scores"] = scores
                results.append(result)
                
            except Exception as e:
                print(f"❌ Erreur traitement ligne: {e}")
                continue
        
        print(f"✅ {len(results)} analyses traitées avec succès")
        return results

    except Exception as e:
        print(f"❌ Erreur récupération analyses: {e}")
        import traceback
        traceback.print_exc()
        return []
    finally:
        conn.close()


def get_comparison_history_fixed(user_id: int = None, limit: int = 100):
    """
    Version corrigée pour les comparaisons
    """
    conn = get_db_connection()
    cursor = conn.cursor()
    
    try:
        # Lister toutes les tables
        cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
        all_tables = [row[0] for row in cursor.fetchall()]
        
        # Chercher les tables de comparaison possibles
        possible_tables = [
            'ai_comparisons',
            'comparisons',
            'comparison',
            'esg_comparisons'
        ]
        
        existing_table = None
        for table in possible_tables:
            if table in all_tables:
                existing_table = table
                break
        
        # Si aucune table standard, chercher 'comparison'
        if not existing_table:
            for table in all_tables:
                if 'comparison' in table.lower():
                    existing_table = table
                    break
        
        if not existing_table:
            print(f"❌ Aucune table de comparaison trouvée dans: {all_tables}")
            return []
        
        print(f"✅ Utilisation de la table: {existing_table}")
        
        # Vérifier le contenu
        cursor.execute(f"SELECT COUNT(*) FROM {existing_table}")
        total_count = cursor.fetchone()[0]
        print(f"📊 Total de comparaisons dans {existing_table}: {total_count}")
        
        # Requête
        base_query = f'''
            SELECT ac.*, r1.filename as filename1, r2.filename as filename2
            FROM {existing_table} ac
            LEFT JOIN reports r1 ON ac.report1_id = r1.id
            LEFT JOIN reports r2 ON ac.report2_id = r2.id
        '''
        
        params = []
        if user_id is not None:
            base_query += " WHERE ac.user_id = ?"
            params.append(user_id)
            print(f"🔍 Filtrage par user_id: {user_id}")
        
        base_query += f" ORDER BY ac.id DESC LIMIT {limit}"
        
        print(f"🔍 Requête comparaisons: {base_query}")
        
        cursor.execute(base_query, params)
        rows = cursor.fetchall()
        
        print(f"📊 Comparaisons trouvées: {len(rows)}")
        
        results = []
        for row in rows:
            try:
                result = {
                    "id": row["id"],
                    "report1_id": row["report1_id"],
                    "report2_id": row["report2_id"],
                    "filename1": row["filename1"] or f"Rapport {row['report1_id']}",
                    "filename2": row["filename2"] or f"Rapport {row['report2_id']}",
                    "comparison_text": row["comparison_text"],
                    "comparison_date": row["comparison_date"] if "comparison_date" in row.keys() else row["created_at"],
                    "model_used": row["model_used"],
                    "tokens_used": row["tokens_used"] if "tokens_used" in row.keys() and row["tokens_used"] is not None else 0
                }
                
                # Scores
                scores = {}
                if "scores_json" in row.keys() and row["scores_json"]:
                    try:
                        scores = json.loads(row["scores_json"])
                    except:
                        pass
                
                result["scores"] = scores
                results.append(result)
                
            except Exception as e:
                print(f"❌ Erreur traitement comparaison: {e}")
                continue
        
        print(f"✅ {len(results)} comparaisons traitées")
        return results
        
    except Exception as e:
        print(f"❌ Erreur récupération comparaisons: {e}")
        import traceback
        traceback.print_exc()
        return []
    finally:
        conn.close()

def get_user_analytics(user_id: int):
    conn = get_db_connection()
    cursor = conn.cursor()
    try:
        cursor.execute("SELECT COUNT(*) FROM reports WHERE user_id = ?", (user_id,))
        total_reports = cursor.fetchone()[0]

        cursor.execute("SELECT COUNT(*) FROM advanced_analyses WHERE user_id = ?", (user_id,))
        total_advanced_analyses = cursor.fetchone()[0]

        cursor.execute('''
            SELECT AVG(json_extract(scores_json, '$.global_score')) as avg_global,
                   AVG(json_extract(scores_json, '$.e1_score')) as avg_e1,
                   AVG(json_extract(scores_json, '$.e2_score')) as avg_e2,
                   AVG(json_extract(scores_json, '$.e3_score')) as avg_e3,
                   AVG(json_extract(scores_json, '$.e4_score')) as avg_e4,
                   AVG(json_extract(scores_json, '$.e5_score')) as avg_e5
            FROM advanced_analyses WHERE user_id = ?
        ''', (user_id,))
        avg_scores_row = cursor.fetchone()
        avg_scores = {
            "global_score": round(avg_scores_row[0] or 0, 1),
            "e1_score": round(avg_scores_row[1] or 0, 1),
            "e2_score": round(avg_scores_row[2] or 0, 1),
            "e3_score": round(avg_scores_row[3] or 0, 1),
            "e4_score": round(avg_scores_row[4] or 0, 1),
            "e5_score": round(avg_scores_row[5] or 0, 1)
        }

        cursor.execute('''
            SELECT DATE(analysis_date) as date,
                   AVG(json_extract(scores_json, '$.global_score')) as daily_avg
            FROM advanced_analyses 
            WHERE user_id = ? 
            GROUP BY DATE(analysis_date)
            ORDER BY date DESC
            LIMIT 30
        ''', (user_id,))
        score_evolution = [{"date": row[0], "score": round(row[1], 1)} for row in cursor.fetchall()]

        return {
            "user_id": user_id,
            "total_reports": total_reports,
            "total_advanced_analyses": total_advanced_analyses,
            "average_scores": avg_scores,
            "score_evolution": score_evolution,
            "generated_at": datetime.now().isoformat()
        }
    except Exception as e:
        print(f"❌ Erreur récupération analytics: {e}")
        raise
    finally:
        conn.close()
